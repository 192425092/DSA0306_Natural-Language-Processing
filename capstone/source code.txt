<script>

function cleanText(text) {

    return text
        .toLowerCase()
        .replace(/[^\w\s]/g, "")
        .split(/\s+/)
        .filter(word => word.length > 2);

}


function calculateSimilarity(original, submitted) {

    let originalWords = cleanText(original);
    let submittedWords = cleanText(submitted);

    let originalSet = new Set(originalWords);
    let submittedSet = new Set(submittedWords);

    let common = 0;

    submittedSet.forEach(word => {

        if (originalSet.has(word)) {
            common++;
        }

    });

    let union = new Set([
        ...originalSet,
        ...submittedSet
    ]).size;

    if (union === 0) {
        return 0;
    }

    return Math.round((common / union) * 100);

}


function findMatches(original, submitted) {

    let originalWords = cleanText(original);
    let submittedWords = cleanText(submitted);

    let commonWords = submittedWords.filter(
        word => originalWords.includes(word)
    );

    let uniqueWords = [...new Set(commonWords)];

    return uniqueWords.slice(0, 8);

}


function analyzeText() {

    let original = document.getElementById("originalText").value.trim();
    let submitted = document.getElementById("submittedText").value.trim();

    if (!original || !submitted) {

        alert("Please enter both the original text and submitted text.");

        return;

    }

    let similarity = calculateSimilarity(
        original,
        submitted
    );


    /*
       Demo classification logic

       High similarity = possible plagiarism
       Medium similarity = possible paraphrasing
       Low similarity = mostly original
    */

    let plagiarism = 0;
    let paraphrase = 0;
    let originality = 0;

    if (similarity >= 75) {

        plagiarism = similarity;
        paraphrase = Math.max(0, similarity - 15);
        originality = 100 - plagiarism;

    }

    else if (similarity >= 40) {

        paraphrase = similarity;
        plagiarism = Math.round(similarity * 0.35);
        originality = 100 - paraphrase;

    }

    else {

        originality = 100 - similarity;
        paraphrase = Math.round(similarity * 0.5);
        plagiarism = Math.round(similarity * 0.2);

    }


    document.getElementById("similarity").innerText =
        similarity + "%";

    document.getElementById("plagiarism").innerText =
        plagiarism + "%";

    document.getElementById("paraphrase").innerText =
        paraphrase + "%";

    document.getElementById("originality").innerText =
        originality + "%";


    let bar = document.getElementById("similarityBar");

    bar.style.width = similarity + "%";

    bar.innerText = similarity + "%";


    let status = document.getElementById("status");

    if (similarity >= 75) {

        status.innerText =
            "High Similarity - Possible Plagiarism";

        status.className =
            "status plagiarized";

    }

    else if (similarity >= 40) {

        status.innerText =
            "Moderate Similarity - Possible Paraphrasing";

        status.className =
            "status paraphrased";

    }

    else {

        status.innerText =
            "Low Similarity - Mostly Original";

        status.className =
            "status original";

    }


    updateMatches(original, submitted);

    updateReport(
        similarity,
        plagiarism,
        paraphrase,
        originality
    );

}


function updateMatches(original, submitted) {

    let matches = findMatches(
        original,
        submitted
    );

    let table = document.getElementById("matches");

    table.innerHTML = "";


    if (matches.length === 0) {

        table.innerHTML =
            "<tr><td colspan='3'>No significant matching words detected.</td></tr>";

        return;

    }


    matches.forEach(word => {

        let row = document.createElement("tr");

        row.innerHTML = `

            <td>${word}</td>

            <td>${word}</td>

            <td>Semantic / Keyword Match</td>

        `;

        table.appendChild(row);

    });

}


function updateReport(
    similarity,
    plagiarism,
    paraphrase,
    originality
) {

    let report =
        document.getElementById("report");

    let similarityText =
        similarity >= 75
        ? "Very high semantic similarity"
        : similarity >= 40
        ? "Moderate semantic similarity"
        : "Low semantic similarity";


    let plagiarismText =
        plagiarism >= 75
        ? "Strong plagiarism indication"
        : plagiarism >= 40
        ? "Potential copied content"
        : "Low plagiarism indication";


    let paraphraseText =
        paraphrase >= 40
        ? "Possible paraphrased content"
        : "Low paraphrase indication";


    let originalityText =
        originality >= 70
        ? "Content appears mostly original"
        : "Content requires further review";


    report.innerHTML = `

        <tr>
            <td>Semantic Similarity</td>
            <td>${similarity}%</td>
            <td>${similarityText}</td>
        </tr>

        <tr>
            <td>Plagiarism</td>
            <td>${plagiarism}%</td>
            <td>${plagiarismText}</td>
        </tr>

        <tr>
            <td>Paraphrasing</td>
            <td>${paraphrase}%</td>
            <td>${paraphraseText}</td>
        </tr>

        <tr>
            <td>Originality</td>
            <td>${originality}%</td>
            <td>${originalityText}</td>
        </tr>

    `;

}
